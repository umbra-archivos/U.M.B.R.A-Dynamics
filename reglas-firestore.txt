// REGLAS DE FIRESTORE: pégalas en Firebase > Firestore Database > Reglas > Publicar
// - Cualquiera puede ENVIAR un archivo, pero siempre queda sin aprobar (aprobado:false).
// - Cualquiera puede LEER solo los archivos que tú hayas aprobado (aprobado:true).
// - Nadie desde la web puede editar ni borrar. Tú lo haces desde la consola de Firebase.
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /otrosArchivos/{id} {
      allow read: if resource.data.aprobado == true;
      allow create: if request.resource.data.keys().hasOnly(['nombre', 'imagen', 'lore', 'aprobado', 'creado'])
        && request.resource.data.aprobado == false
        && request.resource.data.nombre is string
        && request.resource.data.nombre.size() > 0
        && request.resource.data.nombre.size() <= 60
        && request.resource.data.imagen is string
        && request.resource.data.imagen.size() <= 500
        && (request.resource.data.imagen == '' || request.resource.data.imagen.matches('^https://.*'))
        && request.resource.data.lore is string
        && request.resource.data.lore.size() > 0
        && request.resource.data.lore.size() <= 4000
        && request.resource.data.creado is number;
      allow update, delete: if false;
    }
  }
}
